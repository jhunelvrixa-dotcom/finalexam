Java Number Guessing Game

This is a sample Java console program. Replace it with your actual CC102 Performance Task game source code. 


import java.util.Scanner;
import java.util.Random;

public class Main {
    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        Random random = new Random();

        int answer = random.nextInt(10) + 1;
        int guess;

        System.out.println("NUMBER GUESSING GAME");
        System.out.println("Guess a number from 1 to 10:");

        guess = input.nextInt();

        if (guess == answer) {
            System.out.println("Congratulations! Correct!");
        } else {
            System.out.println("Sorry! The correct number was "
                    + answer);
        }

        input.close();
    }
}
